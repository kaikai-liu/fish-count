    // src/routes/+page.server.ts — Home (today's per-boat counts)
    // Sources:
    //   .planning/phases/02-browse-trip-picker-trends/02-CONTEXT.md §D-03, D-20, D-21, D-29
    //   .planning/phases/02-browse-trip-picker-trends/02-RESEARCH.md §Pattern 1
    import type { PageServerLoad } from './$types';
    import { getDb } from '$lib/db/client';
    import {
      getRowsForDate,
      distinctTripTypes,
      distinctLandings,
      distinctSpecies
    } from '$lib/db/queries/browse';
    import { latestSuccessOrEmpty } from '$lib/db/scrapeRuns';
    import { today, toPtTimeLabel } from '$lib/shared/dates';
    import { parseHomeFilters } from '$lib/shared/urlState';

    export const load: PageServerLoad = async ({ url, setHeaders, locals }) => {
      const db = getDb();
      const todayStr = today();

      // D-29: today's page caches 60s (still reporting)
      setHeaders({ 'cache-control': 'public, max-age=60' });

      // Parse filters; absent searchParams = no filtering applied
      const filtersResult = parseHomeFilters(url.searchParams);
      const filters = 'error' in filtersResult ? {} : filtersResult;

      // Get today's rows (filters applied client-side via Svelte for now;
      // if filter shape grows, push into a parametrized query in browse.ts)
      let rows = getRowsForDate(db, todayStr);
      if ('tripType' in filters && filters.tripType) {
        rows = rows.filter((r) => r.trip_type === filters.tripType);
      }
      if ('landing' in filters && filters.landing) {
        rows = rows.filter((r) => r.landing_name === filters.landing);
      }
      if ('species' in filters && filters.species) {
        rows = rows.filter((r) => r.species === filters.species);
      }

      const lastScrape = latestSuccessOrEmpty(db);
      const tripTypes = distinctTripTypes(db);
      const landings = distinctLandings(db);
      const speciesList = distinctSpecies(db);

      locals.logger?.info({ msg: 'home_loaded', date: todayStr, rowCount: rows.length });

      return {
        rows,
        date: todayStr,
        isProvisional: true, // D-20: today is always provisional
        lastScrapedLabel: lastScrape ? toPtTimeLabel(lastScrape.finished_at) : null,
        filters,
        filterOptions: { tripTypes, landings, speciesList }
      };
    };
    ```

    import type { PageServerLoad } from './$types';
    import { error } from '@sveltejs/kit';
    import { getDb } from '$lib/db/client';
    import {
      getRowsForDate,
      distinctTripTypes,
      distinctLandings,
      distinctSpecies
    } from '$lib/db/queries/browse';
    import { latestSuccessOrEmpty } from '$lib/db/scrapeRuns';
    import { today, toPtTimeLabel, addDays, clampDate, isPast, isToday } from '$lib/shared/dates';

    const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;

    export const load: PageServerLoad = async ({ params, url, setHeaders, locals }) => {
      const requestedDate = params.date;

      // T-02-14: regex-validate before any further use
      if (!DATE_RE.test(requestedDate)) {
        throw error(404, 'Invalid date format. Use YYYY-MM-DD.');
      }
      // Sanity: not all 8-digit YYYY-MM-DD strings are valid calendars; check via Date parse
      const [y, m, d] = requestedDate.split('-').map(Number);
      const probe = new Date(Date.UTC(y, m - 1, d));
      if (probe.getUTCFullYear() !== y || probe.getUTCMonth() !== m - 1 || probe.getUTCDate() !== d) {
        throw error(404, 'Invalid calendar date.');
      }

      const db = getDb();

      // Compute dataset bounds for clamping prev/next + the requested date itself
      // Use the existing per-table repo to get min source_date — if no rows exist, fall back to today()
      // We can derive bounds from queries/browse helpers if exposed; otherwise inline a tiny query is forbidden (DAL boundary).
      // PATTERN: a bounds helper in queries/browse.ts (Plan 02-01) — if not present, use clampDate to today() as upper bound + a sentinel oldest like '1970-01-01' as lower bound (the DAL's getRowsForDate handles empty-dates safely).
      // For Phase 2-03 simplicity: clamp upper bound to today() (no future dates beyond today); lower bound deferred to dataset min via a future helper.
      const minDate = '1970-01-01';
      const maxDate = today();
      const dateClamped = clampDate(requestedDate, minDate, maxDate);

      // If clamped, redirect to the canonical URL via SvelteKit `redirect` — but for simplicity, just render the clamped date.
      // (A redirect would change URL; rendering with notice is also acceptable for v1.)

      // D-29: past dates cache 1 day; today's-via-/date caches 60s same as /
      const isTodayDate = isToday(dateClamped);
      setHeaders({
        'cache-control': isTodayDate ? 'public, max-age=60' : 'public, max-age=86400'
      });

      const filtersResult = parseDateFilters(url.searchParams);
      const filters = 'error' in filtersResult ? {} : filtersResult;

      let rows = getRowsForDate(db, dateClamped);
      if ('tripType' in filters && filters.tripType) {
        rows = rows.filter((r) => r.trip_type === filters.tripType);
      }
      if ('landing' in filters && filters.landing) {
        rows = rows.filter((r) => r.landing_name === filters.landing);
      }
      if ('species' in filters && filters.species) {
        rows = rows.filter((r) => r.species === filters.species);
      }

      const prevDate = addDays(dateClamped, -1);
      const nextDate = addDays(dateClamped, 1);
      const prevDisabled = prevDate < minDate;
      const nextDisabled = nextDate > maxDate;

      const lastScrape = latestSuccessOrEmpty(db);

      locals.logger?.info({ msg: 'date_loaded', date: dateClamped, rowCount: rows.length });

      return {
        rows,
        date: dateClamped,
        isProvisional: isTodayDate,
        lastScrapedLabel: lastScrape ? toPtTimeLabel(lastScrape.finished_at) : null,
        filters,
        filterOptions: {
          tripTypes: distinctTripTypes(db),
          landings: distinctLandings(db),
          speciesList: distinctSpecies(db)
        },
        nav: {
          prevDate,
          nextDate,
          prevDisabled,
          nextDisabled,
          minDate,
          maxDate
        }
      };
    };
    ```
