```typescript
// STO-04 static enforcement: no YYYY-MM-DD production outside src/lib/shared/dates.ts
import { describe, it, expect } from 'vitest';
import { execSync } from 'node:child_process';

describe('STO-04 date boundary (static grep enforcement)', () => {
  it('no .toISOString().slice(0,10) or .split("T")[0] outside src/lib/shared/dates.ts', () => {
    // These two patterns are the common "extract YYYY-MM-DD from a Date" idioms.
    // Any file outside src/lib/shared/dates.ts using them likely bypasses the
    // single-date-producer rule.
    const patterns = [
      'toISOString\\(\\)\\.slice\\(0',
      '\\.split\\([\\"\']T[\\"\']\\)\\[0\\]'
    ];
    const scopes = ['src/lib/scraper', 'src/lib/db', 'src/lib/server', 'src/lib/alerts', 'scripts'];

    for (const pattern of patterns) {
      let out = '';
      try {
        out = execSync(
          `grep -rnE "${pattern}" ${scopes.join(' ')} 2>/dev/null || true`,
          { encoding: 'utf8' }
        );
      } catch {
        out = '';
      }
      const violations = out
        .split('\n')
        .filter(line => line.trim().length > 0)
        // Allow comments (// ...) to reference these patterns in documentation
        .filter(line => !/^[^:]+:\d+:\s*\/\//.test(line));
      expect(
        violations,
        `Pattern /${pattern}/ must not appear outside src/lib/shared/dates.ts. Violations:\n${violations.join('\n')}`
      ).toEqual([]);
    }
  });

  it('no `const today = new Date()` date-now idiom in src/lib/scraper or scripts', () => {
    // A file calling `new Date()` to produce a YYYY-MM-DD should instead import
    // today() from src/lib/shared/dates.ts. This test flags the raw idiom.
    // Allowed: `new Date().toISOString()` for timestamps (NOT date-only strings).
    // Restricted: `new Date()` used in a variable named `today` / `date` that
    // then produces a YYYY-MM-DD string without going through dates.ts.
    //
    // Because static detection of "this Date flows into a YYYY-MM-DD" is hard,
    // this test uses a heuristic: flag `= new Date()` followed within 3 lines
    // by an `.slice(0, 10)` or `.split('T')` call. In practice, just re-run
    // the first test's patterns — they cover the YYYY-MM-DD extraction.
    expect(true).toBe(true);  // placeholder: first test already covers the hazard
  });
});
```

Note: the second test is intentionally a no-op placeholder because the first test's regex already catches the actual hazard (YYYY-MM-DD extraction from a raw Date). Document this in an inline comment so the test's intent is clear.

The enumerateDates helper in Plan 01-01 produces synthetic date strings from input date strings, which is arguably NOT "today/now production" — but to keep the grep-based enforcement simple, avoid the `.slice(0,10)` idiom anywhere in `src/lib/db/`.

  ```typescript
  function enumerateDates(from: string, to: string): string[] {
    const result: string[] = [];
    let cur = from;
    while (cur <= to) {
      result.push(cur);
      cur = addOneDay(cur);  // pure string arithmetic
    }
    return result;
  }
  function addOneDay(s: string): string {
    // YYYY-MM-DD + 1 day. Handles month/year rollover via Date object,
    // BUT the Date object is constructed from a parameterized input, not `new Date()`,
    // so it's not "now" production. The restricted pattern is .toISOString().slice(0,10).
    const [y, m, d] = s.split('-').map(Number);
    const dt = new Date(Date.UTC(y, m - 1, d + 1));
    const yy = dt.getUTCFullYear();
    const mm = String(dt.getUTCMonth() + 1).padStart(2, '0');
    const dd = String(dt.getUTCDate()).padStart(2, '0');
    return `${yy}-${mm}-${dd}`;
  }
  ```
  This is allowed because it neither uses `new Date()` without arguments NOR uses the `.toISOString().slice(0,10)` pattern. Document in a comment.

```typescript
import { describe, it, expect, beforeEach, afterEach, vi } from 'vitest';
import { mkdtempSync, rmSync, readFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { stubFetch, restoreFetch } from '../../helpers/fetch-stub';

const TYPICAL = readFileSync('tests/fixtures/scraper/2024-08-15-typical.html', 'utf8');

describe('Phase 1 end-to-end integration', () => {
  let tmp: string;
  let originalEnv: Record<string, string | undefined>;
  let originalArgv: string[];

  beforeEach(() => {
    tmp = mkdtempSync(join(tmpdir(), 'fc-e2e-'));
    originalEnv = {
      FIRST_SCRAPE_OK: process.env.FIRST_SCRAPE_OK,
      SCRAPER_ENABLED: process.env.SCRAPER_ENABLED,
      DB_PATH: process.env.DB_PATH,
      SNAPSHOT_DIR: process.env.SNAPSHOT_DIR,
      SCRAPE_LOCK_PATH: process.env.SCRAPE_LOCK_PATH
    };
    process.env.FIRST_SCRAPE_OK = 'true';
    process.env.SCRAPER_ENABLED = 'true';
    process.env.DB_PATH = join(tmp, 'e2e.sqlite3');
    process.env.SNAPSHOT_DIR = join(tmp, 'snapshots');
    process.env.SCRAPE_LOCK_PATH = join(tmp, 'scrape.lock');
    originalArgv = process.argv;
    vi.resetModules();
  });

  afterEach(async () => {
    restoreFetch();
    try {
      const { closeDb } = await import('../../../src/lib/db/client');
      closeDb();
    } catch { /* ignore */ }
    for (const [k, v] of Object.entries(originalEnv)) {
      if (v === undefined) delete process.env[k];
      else process.env[k] = v;
    }
    process.argv = originalArgv;
    rmSync(tmp, { recursive: true, force: true });
    vi.restoreAllMocks();
  });

  it('scenario 1: 3-date backfill writes 3 success ledger rows + catch_reports', async () => {
    process.argv = ['node', 'scripts/backfill.ts', '--from', '2024-08-13', '--to', '2024-08-15', '--quiet'];
    stubFetch([{ body: TYPICAL }, { body: TYPICAL }, { body: TYPICAL }]);

    const { main } = await import('../../../scripts/backfill');
    const code = await main();
    expect(code).toBe(0);

    const { getDb } = await import('../../../src/lib/db/client');
    const db = getDb();
    const ledgerCount = (db.prepare("SELECT COUNT(*) as c FROM scrape_runs WHERE outcome='success'").get() as {c:number}).c;
    expect(ledgerCount).toBe(3);

    const reportCount = (db.prepare('SELECT COUNT(*) as c FROM catch_reports').get() as {c:number}).c;
    expect(reportCount).toBeGreaterThan(0);
  }, 20000);

  it('scenario 2: re-running same range with D-12 resume skips all dates (no fetch)', async () => {
    // Run 1: seed the ledger
    process.argv = ['node', 'scripts/backfill.ts', '--from', '2024-08-13', '--to', '2024-08-15', '--quiet'];
    stubFetch([{ body: TYPICAL }, { body: TYPICAL }, { body: TYPICAL }]);
    const { main } = await import('../../../scripts/backfill');
    await main();

    // Reset fetch stub to track zero calls
    restoreFetch();
    const fetchMock = stubFetch({ body: TYPICAL });

    // Run 2: same range
    vi.resetModules();
    const { main: main2 } = await import('../../../scripts/backfill');
    process.argv = ['node', 'scripts/backfill.ts', '--from', '2024-08-13', '--to', '2024-08-15', '--quiet'];
    const code2 = await main2();

    expect(code2).toBe(0);
    // Expect zero boats.php fetches since all 3 dates have outcome=success
    expect(fetchMock.mock.calls.filter(c => String(c[0]).includes('boats.php')).length).toBe(0);
  }, 30000);

  it('scenario 3: mixed success + http_error → only failed date retried on rerun', async () => {
    process.argv = ['node', 'scripts/backfill.ts', '--from', '2024-08-13', '--to', '2024-08-15', '--quiet'];
    // First date: 200 with fixture. Second date: 503 repeatedly (exhausts retries → http_error). Third: 200.
    // p-retry will try 503 up to 4 times (initial + 3 retries), so we need enough stub responses.
    const responses = [
      { body: TYPICAL },                                   // 2024-08-13 success
      { status: 503, body: 'server error' },               // 2024-08-14 attempt 1
      { status: 503, body: 'server error' },               // attempt 2
      { status: 503, body: 'server error' },               // attempt 3
      { status: 503, body: 'server error' },               // attempt 4 — gives up, http_error
      { body: TYPICAL }                                    // 2024-08-15 success
    ];
    stubFetch(responses);

    const { main } = await import('../../../scripts/backfill');
    const code = await main();
    expect(code).toBe(0);  // CLI treats http_error as non-halting

    const { getDb } = await import('../../../src/lib/db/client');
    const db = getDb();
    const byOutcome = db.prepare('SELECT outcome, COUNT(*) as c FROM scrape_runs GROUP BY outcome').all() as Array<{outcome:string;c:number}>;
    const asMap = Object.fromEntries(byOutcome.map(r => [r.outcome, r.c]));
    expect(asMap['success']).toBe(2);
    expect(asMap['http_error']).toBe(1);
  }, 45000);
});
```

Note: Scenario 3 takes longer because p-retry waits between attempts (up to 8s on the 4th retry). Inside tests, p-retry's default `minTimeout: 1000` with `factor: 2` means retries sleep ~1s, 2s, 4s — ~7s total per errored date. The 45s test timeout accommodates this.
