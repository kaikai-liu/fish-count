<objective>
Build the trip picker route — the centerpiece of Phase 2 product value. User picks a target date + species + trip type (required) → server returns ranked boats by weighted fish/angler + 30-day calendar heatmap + per-row "Why this boat?" panel data. n<5 boats stay visible with low-data badge. Heatmap cells with n<5 render gray via ECharts itemStyle override.

Heatmap option construction is factored into a pure exported helper (`src/routes/picker/heatmapOption.ts::buildHeatmapOption`) so the n<5 → gray override and viridis-stop logic are unit-testable without DOM.

Every credibility guardrail converges here: weighted yield not mean-of-ratios (D-08), required trip type (TRP-05/D-10), n<5 flagged not hidden (TRP-07), gray heatmap cells (TRP-09), per-angler framing (TRP-04).     // src/routes/picker/+page.server.ts — Trip picker (TRP-01..09)
    ```typescript
    // Sources:
    //   .planning/phases/02-browse-trip-picker-trends/02-CONTEXT.md §D-08, D-09, D-10, D-11, D-13, D-14, D-15, D-28, D-29
    //   .planning/phases/02-browse-trip-picker-trends/02-RESEARCH.md §Pitfall 2 (TRP-05 enforcement)
    import type { PageServerLoad } from './$types';
    import { getDb } from '$lib/db/client';
    import {
      rankBoatsForQuery,
      heatmapForQuery,
      type RankedBoat,
      type HeatmapCell
    } from '$lib/db/queries/tripPicker';
    import { distinctTripTypes, distinctSpecies, mostCommonTripType } from '$lib/db/queries/browse';
    import { latestSuccessOrEmpty } from '$lib/db/scrapeRuns';
    import { today, toPtTimeLabel, addDays } from '$lib/shared/dates';
    import { parsePickerFilters, type PickerFilters } from '$lib/shared/urlState';

    export const load: PageServerLoad = async ({ url, setHeaders, locals }) => {
      const db = getDb();

      // D-29: picker caches 5 min (input-derived)
      setHeaders({ 'cache-control': 'public, max-age=300' });

      const filterOptions = {
        tripTypes: distinctTripTypes(db),
        speciesList: distinctSpecies(db),
        defaultTripType: mostCommonTripType(db),  // D-10: default for the trip-type select
        defaultDate: today()
      };
      const lastScrape = latestSuccessOrEmpty(db);
      const lastScrapedLabel = lastScrape ? toPtTimeLabel(lastScrape.finished_at) : null;

      // T-02-19: TRP-05 enforcement at the SERVER level — URL bypass returns guidance, not rankings.
      const parseResult = parsePickerFilters(url.searchParams);

      if ('error' in parseResult || !url.searchParams.get('tripType')) {
        // Empty URL or invalid params → no rankings; render filter form only.
        locals.logger?.info({ msg: 'picker_guidance', reason: 'missing_or_invalid_filters' });
        return {
          filters: null,
          guidance: 'Pick a date, target species, and trip type to start.',
          rankings: null,
          heatmap: null,
          why: null,
          filterOptions,
          lastScrapedLabel
        };
      }

      const filters = parseResult as PickerFilters;

      // Compute window:
      //   range mode: explicit fromDate / toDate
      //   single mode: target date ± windowDays
      let fromDate: string;
      let toDate: string;
      if (filters.rangeMode && filters.fromDate && filters.toDate) {
        fromDate = filters.fromDate;
        toDate = filters.toDate;
      } else {
        fromDate = addDays(filters.date, -filters.windowDays);
        toDate = addDays(filters.date, filters.windowDays);
      }

      const rankings = rankBoatsForQuery(db, {
        fromDate,
        toDate,
        species: filters.species,
        tripType: filters.tripType
      });

      // Heatmap: 30-cell window starting at filters.date (or filters.fromDate in range mode).
      const heatmapStart = filters.rangeMode && filters.fromDate ? filters.fromDate : filters.date;
      const heatmapEnd = addDays(heatmapStart, 29);
      const presentCells = heatmapForQuery(db, {
        fromDate: heatmapStart,
        toDate: heatmapEnd,
        species: filters.species,
        tripType: filters.tripType
      });

      // Gap-fill: build all 30 dates; if a date is missing from presentCells, emit { date, value: null, n: 0 }.
      const presentMap = new Map(presentCells.map((c) => [c.date, c]));
      const heatmap: HeatmapCell[] = [];
      for (let i = 0; i < 30; i++) {
        const d = addDays(heatmapStart, i);
        const found = presentMap.get(d);
        heatmap.push(found ?? { date: d, value: null, n: 0 });
      }

      // Why-this-boat data (Phase 2 pragmatic shortcut: per-boat best-day uses last_trip_date + avg_per_angler from rankings).
      const why: Record<number, { species: string; tripType: string; windowStart: string; windowEnd: string; bestDay: { date: string; value: number } | null }> = {};
      for (const r of rankings) {
        why[r.boat_id] = {
          species: filters.species,
          tripType: filters.tripType,
          windowStart: fromDate,
          windowEnd: toDate,
          bestDay: r.last_trip_date && r.avg_per_angler !== null
            ? { date: r.last_trip_date, value: r.avg_per_angler }
            : null
        };
      }

      locals.logger?.info({
        msg: 'picker_loaded',
        species: filters.species,
        tripType: filters.tripType,
        rankingCount: rankings.length,
        windowDays: filters.windowDays
      });

      return {
        filters,
        guidance: null,
        rankings,
        heatmap,
        why,
        windowStart: fromDate,
        windowEnd: toDate,
        heatmapRange: { from: heatmapStart, to: heatmapEnd },
        filterOptions,
        lastScrapedLabel
      };
    };
    ```

    ```typescript
    // src/routes/picker/heatmapOption.ts — Pure ECharts option builder for the
    // 30-day calendar heatmap on /picker. Factored out of +page.svelte so the
    // n<5 → gray override (D-14) is unit-testable without DOM/JSDOM.
    //
    // Input: gap-filled HeatmapCell[] (length 30) + range {from, to} from loader.
    // Output: EChartsOption suitable for Chart.svelte (which dynamic-imports echarts).

    import type { EChartsOption } from 'echarts';
    import type { HeatmapCell } from '$lib/db/queries/tripPicker';
    import {
      FISH_PER_ANGLER_TOOLTIP_UNIT,
      HEATMAP_LEGEND_HIGH
    } from '$lib/copy/metrics';

    /** Insufficient-data cell color (UI-SPEC §Color tokens; matches --color-lowdata-bg). */
    export const HEATMAP_LOWDATA_GRAY = '#e5e7eb';

    /** Viridis 5-stop palette (matches src/app.css @theme --heatmap-0..4). */
    export const VIRIDIS_STOPS = ['#440154', '#3b528b', '#21918c', '#5ec962', '#fde725'];

    export function buildHeatmapOption(
      cells: HeatmapCell[],
      range: { from: string; to: string }
    ): EChartsOption {
      // T-02-23: gray override for n<5 cells (D-14)
      const seriesData = cells.map((c) =>
        c.n < 5
          ? { value: [c.date, c.value ?? 0], itemStyle: { color: HEATMAP_LOWDATA_GRAY } }
          : [c.date, c.value]
      );

      // Max value for visualMap derived only from n>=5 cells (avoid n<5 distorting the scale).
      const validValues = cells
        .filter((c) => c.n >= 5 && c.value !== null)
        .map((c) => c.value as number);
      const maxVal = validValues.length > 0 ? Math.max(1, ...validValues) : 1;

      return {
        tooltip: {
          position: 'top',
          formatter: (params: any) => {
            const cell = cells.find((c) => c.date === params.value[0]);
            if (!cell) return '';
            if (cell.n < 5) return `${cell.date}<br/>low data — n=${cell.n}`;
            return `${cell.date}<br/>${(cell.value ?? 0).toFixed(1)} ${FISH_PER_ANGLER_TOOLTIP_UNIT}<br/>n=${cell.n} ${cell.n === 1 ? 'trip' : 'trips'}`;
          }
        },
        calendar: {
          range: [range.from, range.to],
          cellSize: ['auto', 20],
          yearLabel: { show: false }
        },
        visualMap: {
          min: 0,
          max: maxVal,
          calculable: true,
    ```

    ```typescript
    import { describe, it, expect } from 'vitest';
    import { buildHeatmapOption, HEATMAP_LOWDATA_GRAY, VIRIDIS_STOPS } from '../../../src/routes/picker/heatmapOption';

    describe('buildHeatmapOption', () => {
      const range = { from: '2026-05-01', to: '2026-05-30' };

      it('overrides itemStyle.color to gray for n<5 cells', () => {
        const cells = [
          { date: '2026-05-01', value: 2.5, n: 2 },
          { date: '2026-05-02', value: 3.0, n: 8 }
        ];

        // n>=5 cell stays as a bare tuple
        expect(Array.isArray(data[1])).toBe(true);
        expect(data[1]).toEqual(['2026-05-02', 3.0]);
      });

      it('uses gray fallback (value coerced to 0) when n<5 AND value is null', () => {
        const cells = [{ date: '2026-05-01', value: null, n: 1 }];
        const opt = buildHeatmapOption(cells as any, range);
        const data = (opt.series as any)[0].data as any[];
        expect(data[0]).toEqual({ value: ['2026-05-01', 0], itemStyle: { color: HEATMAP_LOWDATA_GRAY } });
      });

      it('derives visualMap.max from n>=5 cells only (n<5 cells do not distort scale)', () => {
        const cells = [
          { date: '2026-05-01', value: 99,  n: 1 },   // outlier but n<5 → ignored
          { date: '2026-05-02', value: 2.5, n: 8 },
          { date: '2026-05-03', value: 3.0, n: 6 }
        ];
        const opt = buildHeatmapOption(cells as any, range);
        expect((opt.visualMap as any).max).toBe(3.0);
      });

      it('falls back to max=1 when there are no n>=5 cells', () => {
        const cells = [{ date: '2026-05-01', value: 2.5, n: 1 }];
        const opt = buildHeatmapOption(cells as any, range);
        expect((opt.visualMap as any).max).toBe(1);
      });

      it('uses viridis palette stops verbatim', () => {
        const opt = buildHeatmapOption([], range);
        expect((opt.visualMap as any).inRange.color).toEqual(VIRIDIS_STOPS);
        expect(VIRIDIS_STOPS[0]).toBe('#440154');
        expect(VIRIDIS_STOPS[4]).toBe('#fde725');
      });

      it('configures calendar with the supplied range', () => {
    ```
