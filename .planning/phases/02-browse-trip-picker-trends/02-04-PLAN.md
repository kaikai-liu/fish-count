<objective>
Build the trip picker route — the centerpiece of Phase 2 product value. User picks a target date + species + trip type (required) → server returns ranked boats by weighted fish/angler + 30-day calendar heatmap + per-row "Why this boat?" panel data. n<5 boats stay visible with low-data badge. Heatmap cells with n<5 render gray via ECharts itemStyle override.

Purpose: This is the "given a date and species, pick the best boat" workflow at the heart of the product. Every credibility guardrail converges here: weighted yield not mean-of-ratios (D-08), required trip type (TRP-05/D-10), n<5 flagged not hidden (TRP-07), gray heatmap cells (TRP-09), per-angler framing (TRP-04). This route is the per-angler metric's first appearance, which is why CLAUDE.md non-negotiable #4 calls Plan 02-04 the highest-discipline page in Phase 2.

    ```typescript
    // src/routes/picker/+page.server.ts — Trip picker (TRP-01..09)
    // Sources:
    //   .planning/phases/02-browse-trip-picker-trends/02-CONTEXT.md §D-08, D-09, D-10, D-11, D-13, D-14, D-15, D-28, D-29
    //   .planning/phases/02-browse-trip-picker-trends/02-RESEARCH.md §Pitfall 2 (TRP-05 enforcement)
    import type { PageServerLoad } from './$types';
    import { getDb } from '$lib/db/client';
    import {
      rankBoatsForQuery,
      heatmapForQuery,
      type RankedBoat,
      type HeatmapCell
    } from '$lib/db/queries/tripPicker';
    import { distinctTripTypes, distinctSpecies, mostCommonTripType } from '$lib/db/queries/browse';
    import { latestSuccessOrEmpty } from '$lib/db/scrapeRuns';
    import { today, toPtTimeLabel, addDays } from '$lib/shared/dates';
    import { parsePickerFilters, type PickerFilters } from '$lib/shared/urlState';

    export const load: PageServerLoad = async ({ url, setHeaders, locals }) => {
      const db = getDb();

      // D-29: picker caches 5 min (input-derived)
      setHeaders({ 'cache-control': 'public, max-age=300' });

      const filterOptions = {
        tripTypes: distinctTripTypes(db),
        speciesList: distinctSpecies(db),
        defaultTripType: mostCommonTripType(db),
        defaultDate: today()
      };
      const lastScrape = latestSuccessOrEmpty(db);
      const lastScrapedLabel = lastScrape ? toPtTimeLabel(lastScrape.finished_at) : null;

      // T-02-19: TRP-05 enforcement at the SERVER level — URL bypass returns guidance, not rankings.
      const parseResult = parsePickerFilters(url.searchParams);

      if ('error' in parseResult || !url.searchParams.get('tripType')) {
        // Empty URL or invalid params → no rankings; render filter form only.
        locals.logger?.info({ msg: 'picker_guidance', reason: 'missing_or_invalid_filters' });
        return {
          filters: null,
          guidance: 'Pick a date, target species, and trip type to start.',
          rankings: null,
          heatmap: null,
          why: null,
          filterOptions,
          lastScrapedLabel
        };
      }

      const filters = parseResult as PickerFilters;

      // Compute window:
      //   range mode: explicit fromDate / toDate
      //   single mode: target date ± windowDays
      let fromDate: string;
      let toDate: string;
      if (filters.rangeMode && filters.fromDate && filters.toDate) {
        fromDate = filters.fromDate;
        toDate = filters.toDate;
      } else {
        fromDate = addDays(filters.date, -filters.windowDays);
        toDate = addDays(filters.date, filters.windowDays);
      }

      const rankings = rankBoatsForQuery(db, {
        fromDate,
        toDate,
        species: filters.species,
        tripType: filters.tripType
      });

      // Heatmap: 30-cell window starting at filters.date (or filters.fromDate in range mode).
      const heatmapStart = filters.rangeMode && filters.fromDate ? filters.fromDate : filters.date;
      const heatmapEnd = addDays(heatmapStart, 29);
      const presentCells = heatmapForQuery(db, {
        fromDate: heatmapStart,
        toDate: heatmapEnd,
        species: filters.species,
        tripType: filters.tripType
      });

      // Gap-fill: build all 30 dates; if a date is missing from presentCells, emit { date, value: null, n: 0 }.
      const presentMap = new Map(presentCells.map((c) => [c.date, c]));
      const heatmap: HeatmapCell[] = [];
      for (let i = 0; i < 30; i++) {
        const d = addDays(heatmapStart, i);
        const found = presentMap.get(d);
        heatmap.push(found ?? { date: d, value: null, n: 0 });
      }

      // Why-this-boat data: derive best-day-in-window per boat from heatmap (one query, not N).
      // For each boat in rankings, fetch best day with same (species, tripType) — fallback to global heatmap if per-boat heatmap is too expensive.
      // Phase 2 pragmatic shortcut: per-boat best-day is the rank's last_trip_date with avg_per_angler value (already in result).
      // For a true per-boat best-day we'd need a heatmapForBoat — defer to v1.x; keep Phase 2 scope tight.
      const why: Record<number, { species: string; tripType: string; windowStart: string; windowEnd: string; bestDay: { date: string; value: number } | null }> = {};
      for (const r of rankings) {
        why[r.boat_id] = {
          species: filters.species,
          tripType: filters.tripType,
          windowStart: fromDate,
          windowEnd: toDate,
          bestDay: r.last_trip_date && r.avg_per_angler !== null
            ? { date: r.last_trip_date, value: r.avg_per_angler }
            : null
        };
      }

      locals.logger?.info({
        msg: 'picker_loaded',
        species: filters.species,
        tripType: filters.tripType,
        rankingCount: rankings.length,
        windowDays: filters.windowDays
      });

      return {
        filters,
        guidance: null,
        rankings,
        heatmap,
        why,
        windowStart: fromDate,
        windowEnd: toDate,
        heatmapRange: { from: heatmapStart, to: heatmapEnd },
        filterOptions,
        lastScrapedLabel
      };
    };
    ```

    ```svelte
    <script lang="ts">
      import type { PageData } from './$types';
      import { goto } from '$app/navigation';
      import PageHeader from '$lib/components/PageHeader.svelte';
      import FilterBar from '$lib/components/FilterBar.svelte';
      import Chart from '$lib/components/Chart.svelte';
      import BoatCard from '$lib/components/BoatCard.svelte';
      import EmptyState from '$lib/components/EmptyState.svelte';
      import PerAnglerFramingProvider from '$lib/components/PerAnglerFramingProvider.svelte';
      import { serializePickerFilters, type PickerFilters } from '$lib/shared/urlState';
      import type { EChartsOption } from 'echarts';

      let { data }: { data: PageData } = $props();

      // Local form state — initialized from data.filters or sensible defaults.
      let formDate = $state(data.filters?.date ?? data.filterOptions.defaultDate);
      let formSpecies = $state(data.filters?.species ?? '');
      let formTripType = $state(data.filters?.tripType ?? data.filterOptions.defaultTripType ?? '');
      let formWindowDays = $state(data.filters?.windowDays ?? 3);
      let formRangeMode = $state(data.filters?.rangeMode ?? false);
      let formFromDate = $state(data.filters?.fromDate ?? '');
      let formToDate = $state(data.filters?.toDate ?? '');

      function submit() {
        if (!formTripType || !formSpecies || !formDate) {
          return; // Native form validation will catch
        }
        const filters: PickerFilters = {
          date: formDate,
          species: formSpecies,
          tripType: formTripType,
          windowDays: formWindowDays,
          rangeMode: formRangeMode,
          fromDate: formRangeMode ? formFromDate : undefined,
          toDate: formRangeMode ? formToDate : undefined
        };
        const sp = serializePickerFilters(filters);
        goto(`/picker?${sp.toString()}`, { keepFocus: true, replaceState: true, noScroll: true });
      }

      function reset() { goto('/picker', { noScroll: true }); }

      // Heatmap chart option — built from data.heatmap when present.
      const heatmapOption = $derived.by<EChartsOption | null>(() => {
        if (!data.heatmap || !data.heatmapRange) return null;
        const cells = data.heatmap;
        // T-02-23: gray override for n<5 cells (D-14)
        const series = {
          type: 'heatmap' as const,
          coordinateSystem: 'calendar' as const,
          data: cells.map((c) =>
            c.n < 5
              ? { value: [c.date, c.value ?? 0], itemStyle: { color: '#e5e7eb' } }
              : [c.date, c.value]
          )
        };
        const validValues = cells.filter((c) => c.n >= 5 && c.value !== null).map((c) => c.value as number);
        const maxVal = validValues.length > 0 ? Math.max(1, ...validValues) : 1;
        return {
          tooltip: {
            position: 'top',
            formatter: (params: any) => {
              const cell = cells.find((c) => c.date === params.value[0]);
              if (!cell) return '';
              if (cell.n < 5) return `${cell.date}<br/>low data — n=${cell.n}`;
              return `${cell.date}<br/>${(cell.value ?? 0).toFixed(1)} fish/angler<br/>n=${cell.n} ${cell.n === 1 ? 'trip' : 'trips'}`;
            }
          },
          calendar: {
            range: [data.heatmapRange.from, data.heatmapRange.to],
            cellSize: ['auto', 20],
            yearLabel: { show: false }
          },
          visualMap: {
            min: 0,
            max: maxVal,
            calculable: true,
            orient: 'horizontal',
            left: 'center',
            bottom: 0,
            inRange: { color: ['#440154', '#3b528b', '#21918c', '#5ec962', '#fde725'] },
            text: ['high (fish/angler)', 'low']
          },
          series: [series]
        } satisfies EChartsOption;
      });

      const heatmapAriaLabel = $derived(
        data.filters
          ? `30-day calendar heatmap of ${data.filters.species} catch on ${data.filters.tripType} trips, fish per angler, gray cells indicate fewer than 5 trips`
          : 'Calendar heatmap (no data)'
    ```
