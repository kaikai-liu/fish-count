```typescript
#!/usr/bin/env tsx
// scripts/backfill.ts — FishCount resumable historical backfill CLI.
// ING-08 + D-10 + D-12.
//
// Required args:
//   --from YYYY-MM-DD
//   --to   YYYY-MM-DD
// Optional:
//   --resume (default: true)
//   --quiet  (default: false)
//
// Required env vars (read by pipeline internals):
//   FIRST_SCRAPE_OK=true   — per D-21, refuses to run otherwise
//   SCRAPER_ENABLED=true   — OPS-05 kill switch (or unset; fail-open)
//   DB_PATH                — path to SQLite file (default /data/fishcount.sqlite3)
//   SNAPSHOT_DIR           — path to snapshots (default /data/snapshots)
//   SCRAPE_LOCK_PATH       — path to lock file (default /data/scrape.lock)
//
// Invocation: `npm run backfill -- --from 2010-01-01 --to 2026-04-23`
//   npm script wires: "backfill": "tsx scripts/backfill.ts"
//
// NO SvelteKit boot: does NOT import from $app/*, $lib/*, hooks.server.ts, or startup.ts.
// Relative `.ts` imports only at this top-level file (PATTERNS.md §SvelteKit-Alias Boundary).

import { parseArgs } from 'node:util';
import { scrapeDate } from '../src/lib/scraper/pipeline.ts';
import { getDatesToScrape } from '../src/lib/db/scrapeRuns.ts';
import { getDb, closeDb } from '../src/lib/db/client.ts';

const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;

function log(msg: string, quiet: boolean): void {
  if (!quiet) process.stdout.write(msg + '\n');
}

export async function main(): Promise<number> {
  let parsed;
  try {
    parsed = parseArgs({
      options: {
        from:   { type: 'string' },
        to:     { type: 'string' },
        resume: { type: 'boolean', default: true },
        quiet:  { type: 'boolean', default: false }
      },
      strict: true,
      allowPositionals: false
    });
  } catch (err) {
    console.error(`[backfill] arg parse error: ${(err as Error).message}`);
    console.error('Usage: npm run backfill -- --from YYYY-MM-DD --to YYYY-MM-DD [--resume] [--quiet]');
    return 2;
  }
  const { values } = parsed;

  if (!values.from || !values.to) {
    console.error('[backfill] --from and --to required');
    console.error('Usage: npm run backfill -- --from YYYY-MM-DD --to YYYY-MM-DD [--resume] [--quiet]');
    return 2;
  }
  if (!DATE_RE.test(values.from) || !DATE_RE.test(values.to)) {
    console.error(`[backfill] invalid date format; expected YYYY-MM-DD (got from=${values.from}, to=${values.to})`);
    return 2;
  }
  if (values.from > values.to) {
    console.error(`[backfill] --from (${values.from}) must be ≤ --to (${values.to})`);
    return 2;
  }

  const quiet = !!values.quiet;
  const db = getDb();
  const dates = getDatesToScrape(db, { from: values.from, to: values.to, resume: values.resume });
  log(`[backfill] ${dates.length} date(s) to process (resume=${values.resume})`, quiet);

  let total = 0;
  const startTime = Date.now();

  for (const date of dates) {
    const result = await scrapeDate(date, 'cli');
    total += result.rowsIngested;
    log(`[${date}] ${result.rowsIngested} rows / ${total} total / ${result.outcome}`, quiet);
    // If the pipeline returned 'killed' because of FIRST_SCRAPE_OK or kill switch,
    // the operator should know to fix + rerun. Exit non-zero so CI catches it.
    if (result.outcome === 'killed') {
      console.error(`[backfill] halted — outcome=killed (${result.errorMessage ?? 'gate closed'})`);
      closeDb();
      return 1;
    }
  }

  const durationMin = ((Date.now() - startTime) / 60000).toFixed(1);
  console.log(`[backfill] complete: ${dates.length} dates attempted, ${total} total rows, ${durationMin} min`);
  closeDb();
  return 0;
}

main()
  .then((code) => process.exit(code))
  .catch((err: unknown) => {
    console.error(`[backfill] fatal: ${(err as Error).message}`);
    try { closeDb(); } catch { /* ignore */ }
    process.exit(1);
  });
```
