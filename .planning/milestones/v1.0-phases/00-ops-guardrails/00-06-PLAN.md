**Step 1** — Author `scripts/kill-switch-drill.sh` (authored in this task):
```bash
#!/bin/bash
# scripts/kill-switch-drill.sh
# Verifies OPS-05 Success Criterion 1: SCRAPER_ENABLED=false halts next tick.
# Requires flyctl + operator access to the deployed fishcount app.
set -euo pipefail

: "${APP_NAME:=fishcount}"

echo "== OPS-05 Kill Switch Drill =="

echo "[1/5] Flipping kill switch ON (SCRAPER_ENABLED=false)"
fly secrets set -a "$APP_NAME" SCRAPER_ENABLED=false

echo "[2/5] Waiting 90s for machine restart + next tick"
sleep 90

echo "[3/5] Tailing logs — looking for 'heartbeat skipped' with reason=kill_switch_set"
LOGS=$(fly logs -a "$APP_NAME" --no-tail 2>&1 | tail -80)
echo "$LOGS"

if echo "$LOGS" | grep -q "heartbeat skipped"; then
  echo "PASS: next tick observed 'heartbeat skipped' after kill switch set"
else
  echo "FAIL: did not observe 'heartbeat skipped' in last 80 log lines"
  echo "Restoring SCRAPER_ENABLED to allow-scraping state"
  fly secrets unset -a "$APP_NAME" SCRAPER_ENABLED || true
  exit 2
fi

echo "[4/5] Flipping kill switch OFF (unset SCRAPER_ENABLED)"
fly secrets unset -a "$APP_NAME" SCRAPER_ENABLED

echo "[5/5] Waiting 90s for machine restart + next tick, checking for resumed 'heartbeat tick'"
sleep 90
LOGS=$(fly logs -a "$APP_NAME" --no-tail 2>&1 | tail -40)
if echo "$LOGS" | grep -q "heartbeat tick"; then
  echo "PASS: heartbeat tick resumed after kill switch unset"
else
  echo "FAIL: heartbeat tick did not resume after unset"
  exit 3
fi

echo ""
echo "== OPS-05 Kill Switch Drill PASSED =="
```
