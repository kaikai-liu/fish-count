```typescript
// src/lib/scraper/sla.ts
// ING-07 row-count SLA alert. Fires when today's total rows for a date drops
// below 50% of the 7-day rolling baseline (success outcomes only in baseline).
// Decision split per PATTERNS.md §SLA:
//   - `shouldAlert` is PURE (testable without DB or Resend)
//   - `checkSlaAndAlert` is the side-effect wrapper
// Source: RESEARCH.md §SLA lines 812-849; D-23/D-24/D-25.
import { getDb } from '$lib/db/client';
import { computeSlaBaseline, type ScrapeOutcome } from '$lib/db/scrapeRuns';
import { totalRowsForDate } from '$lib/db/catchReports';
import { sendOperatorAlert } from '$lib/alerts/operator';
import { logger } from '$lib/server/logger';

const MIN_BASELINE_FOR_ALERT = 5;  // secondary safeguard: don't alert on trickles
const SLA_THRESHOLD = 0.5;         // < 50% of baseline fires

/**
 * Pure decision function.
 * Returns true iff the row-count SLA should alert.
 * Rules (D-23/D-24/D-25):
 *   - Only outcome='success' can trigger (non-success has its own signals).
 *   - baseline must be ≥ MIN_BASELINE_FOR_ALERT (avoids alerting on insufficient history).
 *   - Alert when todayTotal < SLA_THRESHOLD * baseline.
 */
export function shouldAlert(
  outcome: ScrapeOutcome,
  baseline: number | null,
  todayTotal: number
): boolean {
  if (outcome !== 'success') return false;
  if (baseline === null || baseline < MIN_BASELINE_FOR_ALERT) return false;
  return todayTotal < SLA_THRESHOLD * baseline;
}

/**
 * Side-effect wrapper: query DB, decide, dispatch email.
 * Called exclusively from the scheduler tick (NOT from the backfill CLI —
 * mid-history baselines are meaningless).
 */
export async function checkSlaAndAlert(date: string, outcome: ScrapeOutcome): Promise<void> {
  // Fast path: short-circuit before any DB read if outcome is not success
  if (outcome !== 'success') return;

  const db = getDb();
  const baseline = computeSlaBaseline(db, date);
  const total = totalRowsForDate(db, date);

  if (!shouldAlert(outcome, baseline, total)) {
    logger.info({ msg: 'sla_ok', date, total, baseline });
    return;
  }

  const threshold = (SLA_THRESHOLD * (baseline ?? 0)).toFixed(1);
  const subject = `FishCount: row-count SLA breach for ${date}`;
  const body =
    `Row-count SLA alert\n\n` +
    `Date: ${date}\n` +
    `Today total: ${total} rows\n` +
    `7-day baseline (success outcomes only): ${(baseline ?? 0).toFixed(1)} rows\n` +
    `Threshold (50% of baseline): ${threshold} rows\n\n` +
    `Today total (${total}) is below threshold (${threshold}). Investigate parser/schema drift.`;

  logger.warn({ msg: 'sla_breach', date, total, baseline, threshold });
  await sendOperatorAlert({ subject, body });
}
```
