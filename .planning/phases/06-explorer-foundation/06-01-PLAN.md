```typescript
// src/lib/shared/slug.ts
// Pure slugify + collision-resolution helpers used at boat ingestion (D-13 frozen-at-first-seen).
// CLAUDE.md Architecture Rules: this module issues no SQL; it is consumed by src/lib/db/boats.ts only.

/** Lowercase, hyphenated, ASCII slug. Max 60 chars. Never empty. */
export function slugify(name: string): string {
  return name
    .normalize('NFKD')
    .replace(/[̀-ͯ]/g, '')      // strip combining marks
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '')
    .slice(0, 60) || 'boat';
}

/** Returns first of {base, base-2, base-3, ...} not in `taken`. Deterministic. */
export function uniqueSlug(base: string, taken: Set<string>): string {
  if (!taken.has(base)) return base;
  let n = 2;
  while (taken.has(`${base}-${n}`)) n++;
  return `${base}-${n}`;
}
```

Note the regex char class `[̀-ͯ]` (Unicode combining-marks block) — RESEARCH.md uses literal combining characters; the escape form is portable across editors.

```typescript
function backfillSlugs(db: Database.Database): void {
  const rows = db.prepare(`SELECT id, display_name FROM boats WHERE slug IS NULL ORDER BY id ASC`).all() as Array<{ id: number; display_name: string }>;
  const taken = new Set<string>();
  const upd = db.prepare(`UPDATE boats SET slug = ? WHERE id = ?`);
  const tx = db.transaction((items: Array<{ id: number; display_name: string }>) => {
    for (const row of items) {
      const slug = uniqueSlug(slugify(row.display_name), taken);
      taken.add(slug);
      upd.run(slug, row.id);
    }
  });
  tx(rows);
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Scraper input → DAL | Display names from sandiegofishreports.com are untrusted strings. They flow into `slugify()` and become URL identifiers. |
| Migration → existing DB | Migration runs on a populated SQLite DB at boot. Backfill UPDATE could corrupt data if buggy. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-01 | Tampering | `slugify(name)` accepting hostile display_name (very long, control chars, mixed scripts) | mitigate | `.slice(0, 60)` caps URL length; `[^a-z0-9]+/-` collapses any non-alphanumeric to single hyphen; `|| 'boat'` ensures non-empty; NFKD strips diacritics; output is regex-bounded `^[a-z0-9-]+$` |
| T-06-02 | Tampering | `findBySlug` SQL via crafted slug | mitigate | better-sqlite3 prepared statement with positional `?` parameter; slug field in URL is later validated by Zod regex `^[a-z0-9]+(-[a-z0-9]+)*$` (Plan 02) — but defense-in-depth here means SQL injection impossible regardless |
| T-06-03 | Denial of Service | `slugify` on very long input | mitigate | `.slice(0, 60)` after the regex pass bounds output; input itself is bounded by SvelteKit URL parsing (~8KB) and DB column max (no explicit cap on display_name but realistically <200 chars from source) |
| T-06-04 | Tampering | Backfill UPDATE corrupts existing boats | mitigate | Backfill runs inside `db.transaction(...)` — rollback on any error; `WHERE slug IS NULL` predicate prevents overwriting existing slugs on re-run; `hasSlugColumn` guard prevents re-running ALTER on already-migrated DB |
| T-06-05 | Tampering | UNIQUE INDEX on slug rejects insertion mid-batch | mitigate | `taken: Set<string>` accumulates within `upsertBoatsAndLandings`; `uniqueSlug` returns first non-conflicting variant; ingestion path tested with two same-name boats in one batch |
| T-06-06 | Information Disclosure | Slug leaks PII | accept | Slugs are derived from public boat names already published on sandiegofishreports.com; no PII in source data |
</threat_model>
