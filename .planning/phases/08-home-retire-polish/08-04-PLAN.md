  import { themeToggleAria, THEME_TOGGLE_TOOLTIP } from '$lib/copy/theme';

  let { theme: initial }: { theme: Theme } = $props();

  function cycle() {
    const n = next[theme];
    document.cookie = `${THEME_COOKIE}=${n}; Path=/; SameSite=Lax; Max-Age=31536000`;

// src/lib/shared/dates.ts — bucket-start helpers (RESEARCH §"Granularity Bucketing Math")
import { startOfWeek, startOfMonth, parse, format } from 'date-fns';

export function isoWeekStartFromKey(weekKey: string): string {
  // '2025-W14' → '2025-03-31' (Monday of ISO week 14, 2025)
  const d = parse(weekKey, "RRRR-'W'II", new Date());
  return format(startOfWeek(d, { weekStartsOn: 1 }), 'yyyy-MM-dd');
}

export function monthStartFromKey(monthKey: string): string {
  // '2025-04' → '2025-04-01'
  return `${monthKey}-01`;
}

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → fc_theme cookie → server transformPageChunk | Untrusted cookie value → injected into HTML attribute. The validator (validateTheme) is the trust boundary. |
| Browser → URL granularity param → loader | Untrusted query param. Zod enum validation rejects garbage. |
| Browser → /compare typeahead input | Untrusted typed string. Resolved via JS lookup against server-provided allBoats list (no DB query per keystroke). |
| Loader throw → +error.svelte | Server-controlled boundary; preserved status code; no stack trace leaks to user. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-04-01 | Tampering / XSS | fc_theme cookie injection into HTML attribute | mitigate | `validateTheme` accepts only literal 'auto'|'light'|'dark'; fallback 'auto'. transformPageChunk replaces `data-theme="%fc_theme%"` with the validated value, NEVER the raw cookie. RESEARCH §Pitfall 2 + Task 1 unit test. |
| T-08-04-02 | Tampering | Cookie tampering for session forgery (NOT applicable to fc_theme) | accept | fc_theme is non-security-sensitive (cosmetic only). No HMAC needed. SameSite=Lax + 1-year max-age + Path=/ are appropriate. |
| T-08-04-03 | Tampering | URL granularity param injection | mitigate | Zod enum validation in ExplorerFiltersSchema rejects any value not in {'daily','weekly','monthly'}. Invalid query → Zod throws → caught by SvelteKit error handler → +error.svelte renders friendly page. |
| T-08-04-04 | Information disclosure | Stack traces leaked via error boundary | mitigate | +error.svelte renders only `page.error.message` (already sanitized by SvelteKit) + `page.status`. No raw error object exposed. RESEARCH §Pattern 6. |
| T-08-04-05 | Information disclosure | /compare typeahead leaks all boat names | accept | Boat names are public data (per source site sandiegofishreports.com). Server-provided allBoats list is identical to what the explorer currently exposes. No PII. |
